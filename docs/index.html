document.addEventListener('DOMContentLoaded', function() {
  if (!window.INEG) return;
  var xp = window.INEG.loadXP();
  if (xp.total > 0) {
    var lev = window.INEG.getNiveau(xp.total);
    var next = window.INEG.getNextNiveau(xp.total);
    var el = document.getElementById('av-welcome');
    var txt = document.getElementById('av-welcome-text');
    el.style.display = 'block';
    txt.innerHTML = 'Thomas is <strong>' + lev.emoji + ' ' + lev.name + '</strong> · ' + xp.total + ' XP'
      + (next ? ' · <strong>' + (next.threshold - xp.total) + ' XP</strong> to reach ' + next.name : ' · maximum level reached!');
  }
  var idx = window.INEG.NIVEAUX.indexOf(window.INEG.getNiveau(xp.total));
  document.querySelectorAll('.av-level').forEach(function(c, i) {
    if (i < idx) c.classList.add('unlocked');
    if (i === idx) c.classList.add('current');
  });
});
</script>
